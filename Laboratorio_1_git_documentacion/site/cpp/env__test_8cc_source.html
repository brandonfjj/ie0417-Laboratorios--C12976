24 TEST_F(EnvTest, ReadWrite) {
25  Random rnd(test::RandomSeed());
26 
27  // Get file to use for testing.
28  std::string test_dir;
29  ASSERT_LEVELDB_OK(env_->GetTestDirectory(&test_dir));
30  std::string test_file_name = test_dir + "/open_on_read.txt";
31  WritableFile* writable_file;
32  ASSERT_LEVELDB_OK(env_->NewWritableFile(test_file_name, &writable_file));
33 
34  // Fill a file with data generated via a sequence of randomly sized writes.
35  static const size_t kDataSize = 10 * 1048576;
36  std::string data;
37  while (data.size() < kDataSize) {
38  int len = rnd.Skewed(18); // Up to 2^18 - 1, but typically much smaller
39  std::string r;
40  test::RandomString(&rnd, len, &r);
41  ASSERT_LEVELDB_OK(writable_file->Append(r));
42  data += r;
43  if (rnd.OneIn(10)) {
44  ASSERT_LEVELDB_OK(writable_file->Flush());
45  }
46  }
47  ASSERT_LEVELDB_OK(writable_file->Sync());
48  ASSERT_LEVELDB_OK(writable_file->Close());
49  delete writable_file;
50 
51  // Read all data using a sequence of randomly sized reads.
52  SequentialFile* sequential_file;
53  ASSERT_LEVELDB_OK(env_->NewSequentialFile(test_file_name, &sequential_file));
54  std::string read_result;
55  std::string scratch;
56  while (read_result.size() < data.size()) {
57  int len = std::min<int>(rnd.Skewed(18), data.size() - read_result.size());
58  scratch.resize(std::max(len, 1)); // at least 1 so &scratch[0] is legal
59  Slice read;
60  ASSERT_LEVELDB_OK(sequential_file->Read(len, &read, &scratch[0]));
61  if (len > 0) {
62  ASSERT_GT(read.size(), 0);
63  }
64  ASSERT_LE(read.size(), len);
65  read_result.append(read.data(), read.size());
66  }
67  ASSERT_EQ(read_result, data);
68  delete sequential_file;
69 }
