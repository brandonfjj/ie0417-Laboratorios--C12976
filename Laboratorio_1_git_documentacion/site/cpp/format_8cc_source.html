32 void Footer::EncodeTo(std::string* dst) const {
33  const size_t original_size = dst->size();
34  metaindex_handle_.EncodeTo(dst);
35  index_handle_.EncodeTo(dst);

43 Status Footer::DecodeFrom(Slice* input) {
44  if (input->size() < kEncodedLength) {
45  return Status::Corruption("not an sstable (footer too short)");
46  }
47 
48  const char* magic_ptr = input->data() + kEncodedLength - 8;
49  const uint32_t magic_lo = DecodeFixed32(magic_ptr);
50  const uint32_t magic_hi = DecodeFixed32(magic_ptr + 4);
51  const uint64_t magic = ((static_cast<uint64_t>(magic_hi) << 32) |
52  (static_cast<uint64_t>(magic_lo)));
53  if (magic != kTableMagicNumber) {
54  return Status::Corruption("not an sstable (bad magic number)");
55  }
56 
57  Status result = metaindex_handle_.DecodeFrom(input);
58  if (result.ok()) {
59  result = index_handle_.DecodeFrom(input);
60  }
61  if (result.ok()) {
62  // We skip over any leftover data (just padding for now) in "input"
63  const char* end = magic_ptr + 8;
64  *input = Slice(end, input->data() + input->size() - end);
65  }
66  return result;
67 }

69 Status ReadBlock(RandomAccessFile* file, const ReadOptions& options,
70  const BlockHandle& handle, BlockContents* result) {
71  result->data = Slice();
72  result->cachable = false;
73  result->heap_allocated = false;
74 
75  // Read the block contents as well as the type/crc footer.
76  // See table_builder.cc for the code that built this structure.
77  size_t n = static_cast<size_t>(handle.size());
78  char* buf = new char[n + kBlockTrailerSize];
79  Slice contents;
80  Status s = file->Read(handle.offset(), n + kBlockTrailerSize, &contents, buf);
81  if (!s.ok()) {
82  delete[] buf;
83  return s;
84  }
85  if (contents.size() != n + kBlockTrailerSize) {
86  delete[] buf;
87  return Status::Corruption("truncated block read");
88  }
89 
90  // Check the crc of the type and the block contents
91  const char* data = contents.data(); // Pointer to where Read put the data
92  if (options.verify_checksums) {
93  const uint32_t crc = crc32c::Unmask(DecodeFixed32(data + n + 1));
94  const uint32_t actual = crc32c::Value(data, n + 1);
95  if (actual != crc) {
96  delete[] buf;
97  s = Status::Corruption("block checksum mismatch");
98  return s;
99  }
100  }
101 
102  switch (data[n]) {
103  case kNoCompression:
104  if (data != buf) {
105  // File implementation gave us pointer to some other data.
106  // Use it directly under the assumption that it will be live
107  // while the file is open.
108  delete[] buf;
109  result->data = Slice(data, n);
110  result->heap_allocated = false;
111  result->cachable = false; // Do not double-cache
112  } else {
113  result->data = Slice(buf, n);
114  result->heap_allocated = true;
115  result->cachable = true;
116  }
117 
118  // Ok
119  break;
120  case kSnappyCompression: {
121  size_t ulength = 0;
122  if (!port::Snappy_GetUncompressedLength(data, n, &ulength)) {
123  delete[] buf;
124  return Status::Corruption("corrupted snappy compressed block length");
125  }
126  char* ubuf = new char[ulength];
127  if (!port::Snappy_Uncompress(data, n, ubuf)) {
128  delete[] buf;
129  delete[] ubuf;
130  return Status::Corruption("corrupted snappy compressed block contents");
131  }
132  delete[] buf;
133  result->data = Slice(ubuf, ulength);
134  result->heap_allocated = true;
135  result->cachable = true;
136  break;
137  }
138  case kZstdCompression: {
139  size_t ulength = 0;
140  if (!port::Zstd_GetUncompressedLength(data, n, &ulength)) {
141  delete[] buf;
142  return Status::Corruption("corrupted zstd compressed block length");
143  }
144  char* ubuf = new char[ulength];
145  if (!port::Zstd_Uncompress(data, n, ubuf)) {
146  delete[] buf;
147  delete[] ubuf;
148  return Status::Corruption("corrupted zstd compressed block contents");
149  }
150  delete[] buf;
151  result->data = Slice(ubuf, ulength);
152  result->heap_allocated = true;
153  result->cachable = true;
154  break;
155  }
156  default:
157  delete[] buf;
158  return Status::Corruption("bad block type");
159  }
160 
161  return Status::OK();
162 }
