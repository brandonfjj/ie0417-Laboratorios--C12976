94 void TableBuilder::Add(const Slice& key, const Slice& value) {
95  Rep* r = rep_;
96  assert(!r->closed);
97  if (!ok()) return;
98  if (r->num_entries > 0) {
99  assert(r->options.comparator->Compare(key, Slice(r->last_key)) > 0);
100  }
101 
102  if (r->pending_index_entry) {
103  assert(r->data_block.empty());
104  r->options.comparator->FindShortestSeparator(&r->last_key, key);
105  std::string handle_encoding;
106  r->pending_handle.EncodeTo(&handle_encoding);
107  r->index_block.Add(r->last_key, Slice(handle_encoding));
108  r->pending_index_entry = false;
109  }
110 
111  if (r->filter_block != nullptr) {
112  r->filter_block->AddKey(key);
113  }
114 
115  r->last_key.assign(key.data(), key.size());
116  r->num_entries++;
117  r->data_block.Add(key, value);
118 
119  const size_t estimated_block_size = r->data_block.CurrentSizeEstimate();
120  if (estimated_block_size >= r->options.block_size) {
121  Flush();
122  }
123 }

125 void TableBuilder::Flush() {
126  Rep* r = rep_;
127  assert(!r->closed);
128  if (!ok()) return;
129  if (r->data_block.empty()) return;
130  assert(!r->pending_index_entry);
131  WriteBlock(&r->data_block, &r->pending_handle);
132  if (ok()) {
133  r->pending_index_entry = true;
134  r->status = r->file->Flush();
135  }
136  if (r->filter_block != nullptr) {
137  r->filter_block->StartBlock(r->offset);
138  }
139 }

141 void TableBuilder::WriteBlock(BlockBuilder* block, BlockHandle* handle) {
142  // File format contains a sequence of blocks where each block has:
143  // block_data: uint8[n]
144  // type: uint8
145  // crc: uint32
146  assert(ok());
147  Rep* r = rep_;
148  Slice raw = block->Finish();
149 
150  Slice block_contents;
151  CompressionType type = r->options.compression;
152  // TODO(postrelease): Support more compression options: zlib?
153  switch (type) {
154  case kNoCompression:
155  block_contents = raw;
156  break;
157 
158  case kSnappyCompression: {
159  std::string* compressed = &r->compressed_output;
160  if (port::Snappy_Compress(raw.data(), raw.size(), compressed) &&
161  compressed->size() < raw.size() - (raw.size() / 8u)) {
162  block_contents = *compressed;
163  } else {
164  // Snappy not supported, or compressed less than 12.5%, so just
165  // store uncompressed form
166  block_contents = raw;
167  type = kNoCompression;
168  }
169  break;
170  }
171 
172  case kZstdCompression: {
173  std::string* compressed = &r->compressed_output;
174  if (port::Zstd_Compress(r->options.zstd_compression_level, raw.data(),
175  raw.size(), compressed) &&
176  compressed->size() < raw.size() - (raw.size() / 8u)) {
177  block_contents = *compressed;
178  } else {
179  // Zstd not supported, or compressed less than 12.5%, so just
180  // store uncompressed form
181  block_contents = raw;
182  type = kNoCompression;
183  }
184  break;
185  }
186  }
187  WriteRawBlock(block_contents, type, handle);
188  r->compressed_output.clear();
189  block->Reset();
190 }

192 void TableBuilder::WriteRawBlock(const Slice& block_contents,
193  CompressionType type, BlockHandle* handle) {
194  Rep* r = rep_;
195  handle->set_offset(r->offset);
196  handle->set_size(block_contents.size());
197  r->status = r->file->Append(block_contents);
198  if (r->status.ok()) {
199  char trailer[kBlockTrailerSize];
200  trailer[0] = type;
201  uint32_t crc = crc32c::Value(block_contents.data(), block_contents.size());
202  crc = crc32c::Extend(crc, trailer, 1); // Extend crc to cover block type
203  EncodeFixed32(trailer + 1, crc32c::Mask(crc));
204  r->status = r->file->Append(Slice(trailer, kBlockTrailerSize));
205  if (r->status.ok()) {
206  r->offset += block_contents.size() + kBlockTrailerSize;
207  }
208  }
209 }

213 Status TableBuilder::Finish() {
214  Rep* r = rep_;
215  Flush();
216  assert(!r->closed);
217  r->closed = true;
218 
219  BlockHandle filter_block_handle, metaindex_block_handle, index_block_handle;
220 
221  // Write filter block
222  if (ok() && r->filter_block != nullptr) {
223  WriteRawBlock(r->filter_block->Finish(), kNoCompression,
224  &filter_block_handle);
225  }
226 
227  // Write metaindex block
228  if (ok()) {
229  BlockBuilder meta_index_block(&r->options);
230  if (r->filter_block != nullptr) {
231  // Add mapping from "filter.Name" to location of filter data
232  std::string key = "filter.";
233  key.append(r->options.filter_policy->Name());
234  std::string handle_encoding;
235  filter_block_handle.EncodeTo(&handle_encoding);
236  meta_index_block.Add(key, handle_encoding);
237  }
238 
239  // TODO(postrelease): Add stats and other meta blocks
240  WriteBlock(&meta_index_block, &metaindex_block_handle);
241  }
242 
243  // Write index block
244  if (ok()) {
245  if (r->pending_index_entry) {
246  r->options.comparator->FindShortSuccessor(&r->last_key);
247  std::string handle_encoding;
248  r->pending_handle.EncodeTo(&handle_encoding);
249  r->index_block.Add(r->last_key, Slice(handle_encoding));
250  r->pending_index_entry = false;
251  }
252  WriteBlock(&r->index_block, &index_block_handle);
253  }
254 
255  // Write footer
256  if (ok()) {
257  Footer footer;
258  footer.set_metaindex_handle(metaindex_block_handle);
259  footer.set_index_handle(index_block_handle);
260  std::string footer_encoding;
261  footer.EncodeTo(&footer_encoding);
262  r->status = r->file->Append(footer_encoding);
263  if (r->status.ok()) {
264  r->offset += footer_encoding.size();
265  }
266  }
267  return r->status;
268 }
