27 // Return reverse of "key".
28 // Used to test non-lexicographic comparators.
29 static std::string Reverse(const Slice& key) {
30  std::string str(key.ToString());
31  std::string rev("");
32  for (std::string::reverse_iterator rit = str.rbegin(); rit != str.rend();
33  ++rit) {
34  rev.push_back(*rit);
35  }
36  return rev;
37 }

67 static void Increment(const Comparator* cmp, std::string* key) {
68  if (cmp == BytewiseComparator()) {
69  key->push_back('\0');
70  } else {
71  assert(cmp == &reverse_key_comparator);
72  std::string rev = Reverse(*key);
73  rev.push_back('\0');
74  *key = Reverse(rev);
75  }

483  void TestRandomAccess(Random* rnd, const std::vector<std::string>& keys,
484  const KVMap& data) {
485  static const bool kVerbose = false;
486  Iterator* iter = constructor_->NewIterator();
487  ASSERT_TRUE(!iter->Valid());
488  KVMap::const_iterator model_iter = data.begin();
489  if (kVerbose) std::fprintf(stderr, "---\n");
490  for (int i = 0; i < 200; i++) {
491  const int toss = rnd->Uniform(5);
492  switch (toss) {
493  case 0: {
494  if (iter->Valid()) {
495  if (kVerbose) std::fprintf(stderr, "Next\n");
496  iter->Next();
497  ++model_iter;
498  ASSERT_EQ(ToString(data, model_iter), ToString(iter));
499  }
500  break;
501  }
502 
503  case 1: {
504  if (kVerbose) std::fprintf(stderr, "SeekToFirst\n");
505  iter->SeekToFirst();
506  model_iter = data.begin();
507  ASSERT_EQ(ToString(data, model_iter), ToString(iter));
508  break;
509  }
510 
511  case 2: {
512  std::string key = PickRandomKey(rnd, keys);
513  model_iter = data.lower_bound(key);
514  if (kVerbose)
515  std::fprintf(stderr, "Seek '%s'\n", EscapeString(key).c_str());
516  iter->Seek(Slice(key));
517  ASSERT_EQ(ToString(data, model_iter), ToString(iter));
518  break;
519  }
520 
521  case 3: {
522  if (iter->Valid()) {
523  if (kVerbose) std::fprintf(stderr, "Prev\n");
524  iter->Prev();
525  if (model_iter == data.begin()) {
526  model_iter = data.end(); // Wrap around to invalid value
527  } else {
528  --model_iter;
529  }
530  ASSERT_EQ(ToString(data, model_iter), ToString(iter));
531  }
532  break;
533  }
534 
535  case 4: {
536  if (kVerbose) std::fprintf(stderr, "SeekToLast\n");
537  iter->SeekToLast();
538  if (keys.empty()) {
539  model_iter = data.end();
540  } else {
541  std::string last = data.rbegin()->first;
542  model_iter = data.lower_bound(last);
543  }
544  ASSERT_EQ(ToString(data, model_iter), ToString(iter));
545  break;
546  }
547  }
548  }
549  delete iter;
550  }

577  std::string PickRandomKey(Random* rnd, const std::vector<std::string>& keys) {
578  if (keys.empty()) {
579  return "foo";
580  } else {
581  const int index = rnd->Uniform(keys.size());
582  std::string result = keys[index];
583  switch (rnd->Uniform(3)) {
584  case 0:
585  // Return an existing key
586  break;
587  case 1: {
588  // Attempt to return something smaller than an existing key
589  if (!result.empty() && result[result.size() - 1] > '\0') {
590  result[result.size() - 1]--;
591  }
592  break;
593  }
594  case 2: {
595  // Return something larger than an existing key
596  Increment(options_.comparator, &result);
597  break;
598  }
599  }
600  return result;
601  }
602  }
